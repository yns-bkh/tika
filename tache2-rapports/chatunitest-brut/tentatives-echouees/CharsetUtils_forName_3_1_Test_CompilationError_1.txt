public class CharsetUtils_forName_3_1_Test {

    private CharsetUtils charsetUtils;

    private Method getCharsetICU;

    private Method isSupportedICU;

    @BeforeEach
    public void setUp() throws Exception {
        charsetUtils = new CharsetUtils();
        getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        getCharsetICU.setAccessible(true);
        isSupportedICU.setAccessible(true);
    }

    @Test
    public void testForNameWithNull() {
        assertThrows(IllegalArgumentException.class, () -> {
            charsetUtils.forName(null);
        });
    }

    @Test
    public void testForNameWithInvalidName() {
        assertThrows(IllegalCharsetNameException.class, () -> {
            charsetUtils.forName("invalid");
        });
    }

    @Test
    public void testForNameWithValidName() {
        Charset charset = charsetUtils.forName("UTF-8");
        assertNotNull(charset);
        assertEquals("UTF-8", charset.name());
    }

    @Test
    public void testForNameWithISOName() {
        Charset charset = charsetUtils.forName("iso 8859-1");
        assertNotNull(charset);
        assertEquals("ISO-8859-1", charset.name());
    }

    @Test
    public void testForNameWithCPName() {
        Charset charset = charsetUtils.forName("cp-1252");
        assertNotNull(charset);
        assertEquals("CP1252", charset.name());
    }

    @Test
    public void testForNameWithWinName() {
        Charset charset = charsetUtils.forName("win-1252");
        assertNotNull(charset);
        assertEquals("WINDOWS-1252", charset.name());
    }

    @Test
    public void testForNameWithICUCharset() throws Exception {
        Map<String, Charset> commonCharsets = new HashMap<>();
        commonCharsets.put("UTF-8", Charset.forName("UTF-8"));
        CharsetUtils.COMMON_CHARSETS = commonCharsets;
        Method getCharsetICUMock = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        getCharsetICUMock.setAccessible(true);
        getCharsetICUMock.invoke(null, "UTF-8");
        Charset charset = charsetUtils.forName("UTF-8");
        assertNotNull(charset);
        assertEquals("UTF-8", charset.name());
    }
}
