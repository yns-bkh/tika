package org.apache.tika.io;

import org.apache.tika.io.LookaheadInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.lang.reflect.Field;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_reset_8_1_Test {

    @Mock
    private InputStream inputStream;

    @InjectMocks
    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(inputStream, 10);
    }

    @Test
    public void testReset() throws IOException, IllegalAccessException {
        // Arrange
        int mark = 5;
        lookaheadInputStream.mark(mark);
        // Act
        lookaheadInputStream.reset();
        // Assert
        Field positionField = LookaheadInputStream.class.getDeclaredField("position");
        positionField.setAccessible(true);
        assertEquals(mark, positionField.get(lookaheadInputStream));
    }

    @Test
    public void testReset_WithNoMark() throws IOException {
        // Act & Assert
        assertThrows(IllegalStateException.class, () -> {
            lookaheadInputStream.reset();
        });
    }
}
