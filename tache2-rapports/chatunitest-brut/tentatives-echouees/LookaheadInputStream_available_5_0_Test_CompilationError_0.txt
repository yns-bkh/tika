package org.apache.tika.io;

import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.IOException;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_available_5_0_Test {

    @Mock
    private InputStream stream;

    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(stream, 10);
    }

    @Test
    public void testAvailable() throws IOException {
        when(stream.available()).thenReturn(15);
        assertEquals(5, lookaheadInputStream.available());
    }

    @Test
    public void testAvailableWithNoBufferedData() throws IOException {
        when(stream.available()).thenReturn(0);
        assertEquals(0, lookaheadInputStream.available());
    }

    @Test
    public void testAvailableWithNegativeBufferedData() throws IOException {
        when(stream.available()).thenReturn(-5);
        assertEquals(-5, lookaheadInputStream.available());
    }

    @Test
    public void testAvailableWithBufferedData() throws IOException {
        lookaheadInputStream.buffered = 10;
        lookaheadInputStream.position = 5;
        assertEquals(5, lookaheadInputStream.available());
    }

    @Test
    public void testAvailableWithBufferedDataAndPosition() throws IOException {
        lookaheadInputStream.buffered = 10;
        lookaheadInputStream.position = 10;
        assertEquals(0, lookaheadInputStream.available());
    }

    @Test
    public void testAvailableWithBufferedDataAndPositionNegative() throws IOException {
        lookaheadInputStream.buffered = 10;
        lookaheadInputStream.position = 15;
        assertEquals(-5, lookaheadInputStream.available());
    }
}
