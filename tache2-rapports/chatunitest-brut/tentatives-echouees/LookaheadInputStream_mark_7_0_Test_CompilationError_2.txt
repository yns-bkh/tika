package org.apache.tika.io;

import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.IOException;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_mark_7_0_Test {

    @Mock
    private InputStream mockStream;

    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(mockStream, 10);
    }

    @Test
    public void testMark() throws IOException {
        // Arrange
        int readlimit = 5;
        // Act
        lookaheadInputStream.mark(readlimit);
        // Assert
        assertEquals(0, getPrivateField(lookaheadInputStream, "mark"));
        verify(mockStream, never()).mark(anyInt());
    }

    @Test
    public void testMarkWithStreamMark() throws IOException {
        // Arrange
        int readlimit = 5;
        when(mockStream.markSupported()).thenReturn(true);
        // Act
        lookaheadInputStream.mark(readlimit);
        // Assert
        assertEquals(0, getPrivateField(lookaheadInputStream, "mark"));
        verify(mockStream).mark(readlimit);
    }

    private Object getPrivateField(Object obj, String fieldName) throws Exception {
        Field field = obj.getClass().getDeclaredField(fieldName);
        field.setAccessible(true);
        return field.get(obj);
    }
}
