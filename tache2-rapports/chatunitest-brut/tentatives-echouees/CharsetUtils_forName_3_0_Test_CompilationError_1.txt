public class CharsetUtils_forName_3_0_Test {

    private CharsetUtils charsetUtils;

    private Method getCharsetICU;

    private Method isSupportedICU;

    @BeforeEach
    public void setUp() throws NoSuchMethodException {
        charsetUtils = new CharsetUtils();
        getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        getCharsetICU.setAccessible(true);
        isSupportedICU.setAccessible(true);
    }

    @Test
    public void testForName_withNullName() {
        assertThrows(IllegalArgumentException.class, () -> {
            charsetUtils.forName(null);
        });
    }

    @Test
    public void testForName_withInvalidName() {
        assertThrows(IllegalCharsetNameException.class, () -> {
            charsetUtils.forName("invalid");
        });
    }

    @Test
    public void testForName_withValidName() throws Exception {
        String name = "UTF-8";
        Charset expectedCharset = Charset.forName(name);
        when(getCharsetICU.invoke(null, name)).thenReturn(expectedCharset);
        Charset result = charsetUtils.forName(name);
        assertEquals(expectedCharset, result);
        verify(getCharsetICU).invoke(null, name);
    }

    @Test
    public void testForName_withCommonCharset() throws Exception {
        String name = "UTF-8";
        Charset expectedCharset = Charset.forName(name);
        Map<String, Charset> commonCharsets = new HashMap<>();
        commonCharsets.put(name.toLowerCase(Locale.ENGLISH), expectedCharset);
        when(CharsetUtils.COMMON_CHARSETS.get(name.toLowerCase(Locale.ENGLISH))).thenReturn(expectedCharset);
        Charset result = charsetUtils.forName(name);
        assertEquals(expectedCharset, result);
        verify(CharsetUtils.COMMON_CHARSETS).get(name.toLowerCase(Locale.ENGLISH));
    }

    @Test
    public void testForName_withISONamePattern() throws Exception {
        String name = "ISO 8859-1";
        Charset expectedCharset = Charset.forName("iso-8859-1");
        when(getCharsetICU.invoke(null, "iso-8859-1")).thenReturn(expectedCharset);
        Charset result = charsetUtils.forName(name);
        assertEquals(expectedCharset, result);
        verify(getCharsetICU).invoke(null, "iso-8859-1");
    }

    @Test
    public void testForName_withCPNamePattern() throws Exception {

    }
