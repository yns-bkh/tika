package org.apache.tika.io;

import org.apache.tika.io.LookaheadInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_available_5_1_Test {

    @Mock
    private InputStream inputStream;

    @InjectMocks
    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(inputStream, 10);
    }

    @Test
    public void testAvailable() throws IOException {
        when(inputStream.available()).thenReturn(15);
        when(lookaheadInputStream.buffered).thenReturn(20);
        when(lookaheadInputStream.position).thenReturn(5);
        int result = lookaheadInputStream.available();
        assertEquals(15, result);
        verify(inputStream, times(1)).available();
        verify(lookaheadInputStream, times(1)).getBuffered();
        verify(lookaheadInputStream, times(1)).getPosition();
    }
}
