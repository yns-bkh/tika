package org.apache.tika.io;

import org.apache.tika.io.LookaheadInputStream;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.IOException;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_skip_4_0_Test {

    @Mock
    private InputStream mockStream;

    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() {
        lookaheadInputStream = new LookaheadInputStream(mockStream, 10);
    }

    @Test
    public void testSkip() throws IOException {
        when(mockStream.available()).thenReturn(5);
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(3);
        long result = lookaheadInputStream.skip(7);
        assertEquals(3, result);
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
        verify(mockStream, times(1)).available();
    }

    @Test
    public void testSkipWithAvailable() throws IOException {
        when(mockStream.available()).thenReturn(10);
        long result = lookaheadInputStream.skip(7);
        assertEquals(7, result);
        verify(mockStream, never()).read(any(byte[].class), anyInt(), anyInt());
        verify(mockStream, times(1)).available();
    }

    @Test
    public void testSkipWithBuffered() throws IOException {
        setPrivateField(lookaheadInputStream, "position", 5);
        when(mockStream.available()).thenReturn(5);
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(3);
        long result = lookaheadInputStream.skip(7);
        assertEquals(3, result);
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
        verify(mockStream, times(1)).available();
    }

    @Test
    public void testSkipWithNoBuffered() throws IOException {
        setPrivateField(lookaheadInputStream, "position", 10);
        when(mockStream.available()).thenReturn(5);
        when(mockStream.read(any(byte[].class), anyInt(), anyInt())).thenReturn(3);
        long result = lookaheadInputStream.skip(7);
        assertEquals(3, result);
        verify(mockStream, times(1)).read(any(byte[].class), anyInt(), anyInt());
        verify(mockStream, times(1)).available();
    }

    @Test
    public void testSkipWithNegative() throws IOException {
        long result = lookaheadInputStream.skip(-1);
        assertEquals(0, result);
        verify(mockStream, never()).read(any(byte[].class), anyInt(), anyInt());
        verify(mockStream, never()).available();
    }

    @Test
    public void testSkipWithZero() throws IOException {
        long result = lookaheadInputStream.skip(0);
        assertEquals(0, result);
        verify(mockStream, never()).read(any(byte[].class), anyInt(), anyInt());
        verify(mockStream, never()).available();
    }

    private void setPrivateField(Object obj, String fieldName, Object value) throws Exception {
        Field field = obj.getClass().getDeclaredField(fieldName);
        field.setAccessible(true);
        field.set(obj, value);
    }
}
